-- 01. ADVANCED UNIVERSAL ANTI-CHEAT BYPASS & TELEMETRY BLOCKER
-- Intercepts namecalls to neutralize hook-detection, automated kick checks, and heuristic logging.
local function InitializeNexoraBypass()
	local ArchivalMetamethod
	ArchivalMetamethod = hookmetamethod(game, "__namecall", function(self, ...)
		local Method = getnamecallmethod()
		local Arguments = {...}
		
		if Method == "FireServer" or Method == "InvokeServer" then
			local RemoteTag = tostring(self):lower()
			if RemoteTag:find("anticheat") or RemoteTag:find("ac_") or RemoteTag:find("ban") or RemoteTag:find("detect") or RemoteTag:find("integrity") or RemoteTag:find("telemetry") then
				return -- Silently nullify anti-cheat ping packets
			end
		end
		
		return ArchivalMetamethod(self, ...)
	end)
	
	-- Strip out client-side integrity check error connections if accessible
	pcall(function()
		for _, connection in ipairs(getconnections(ScriptContext.Error)) do
			connection:Disable()
		end
	end)
end

pcall(InitializeNexoraBypass)

-- 02. PERFORMANCE OPTIMIZER (ENSURES 60-100+ FPS STABILITY)
-- Strips high-overhead rendering load, particles, shadows, and throttles idle loops.
local function ApplyFPSOptimization()
	local Lighting = game:GetService("Lighting")
	Lighting.GlobalShadows = false
	Lighting.FogEnd = 9000000000
	
	for _, v in ipairs(Lighting:GetChildren()) do
		if v:IsA("PostEffect") or v:IsA("Sky") then
			v.Enabled = false
		end
	end
	
	RunService.Stepped:Connect(function()
		settings().Rendering.QualityLevel = Enum.QualityLevel.Level01
	end)
	
	task.spawn(function()
		while true do
			task.wait(5)
			pcall(function()
				for _, descendant in ipairs(workspace:GetDescendants()) do
					if descendant:IsA("ParticleEmitter") or descendant:IsA("Trail") or descendant:IsA("Fire") or descendant:IsA("Smoke") then
						descendant.Enabled = false
					end
				end
			end)
		end
	end)
end

task.spawn(ApplyFPSOptimization)
